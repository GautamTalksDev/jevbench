Run file format

Load a certificate.json written by jevbench export-certificate. The exporter copies harness numbers into result and must never recompute them.

{
  "schema": "jevbench.certificate.v1",
  "meta": {
    "run_id", "git_sha", "prereg_commit" (bf03af9 for EXP-1),
    "model_resolved" (response model field, never the alias),
    "baseline_model", "dataset", "serving_path",
    "n_bins", "binning", "seed", "interval_method",
    "coverage_empirical" (from results/coverage.json),
    "synthetic": false
  },
  "result": {
    "delta_ece", "ci_low", "ci_high", "ece_easy", "ece_hard"
    — copied from the harness. NEVER recomputed in the exporter.
  },
  "items": [{
    "id", "stratum", "entropy",
    "human": [entailment, neutral, contradiction],
    "probs": […], "confidence" (verbatim),
    "latency_ms",
    "baseline": { "pick", "latency_ms" }
  }]
  No sentence text.
}
